import React from 'react';
import {AppRegistry,Text,View} from 'react-native';
import {sourceRuleValues} from './src/core/sourceRule';
import {runSourceJavaScript} from './src/core/sourceJavaScript';
import fixture from './source-qunxiaoshuo.json';
function App(){
 const lines=['iOS JavaScript loop declarations'];
 const check=(name,ok)=>lines.push((ok?'PASS ':'FAIL ')+name);
 try{
  const urls=sourceRuleValues('<div class="bookname">第一章 (1/3)</div>',fixture.nextContentUrl,{baseUrl:'https://example.org/book/1.html'});
  check('1321 pagination rule',JSON.stringify(urls)===JSON.stringify(['https://example.org/book/1_2.html','https://example.org/book/1_3.html']));
  check('loop value',runSourceJavaScript('var s=0;for(let i=0;i<4;i++){s+=i;}s','')===6);
  for(const [name,script] of [['closure','var out=[];for(let i=0;i<2;i++){out.push(function(){return i;});}out[0]()'],['outside scope','for(let i=0;i<2;i++){};typeof i'],['uninitialized','for(let i;i<2;i++){}']]){
   let rejected=false;try{runSourceJavaScript(script,'')}catch(error){rejected=true}check(name+' rejected',rejected);
  }
 }catch(error){lines.push('FAIL '+String(error));}
 return <View style={{flex:1,paddingTop:70,paddingHorizontal:15}}><Text>{lines.join('\n')}</Text></View>;
}
AppRegistry.registerComponent('apps',()=>App);
