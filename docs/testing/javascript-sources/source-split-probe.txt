import React from 'react';
import {AppRegistry,Text,View} from 'react-native';
import {sourceRuleText} from './src/core/sourceRule';
import {runSourceJavaScript} from './src/core/sourceJavaScript';
import source from './source-xinbiquge.json';
function App(){
 const lines=['iOS source regex split'];
 const check=(name,actual,expected)=>lines.push((actual===expected?'PASS ':'FAIL ')+name+' '+actual);
 try{
  const html='<div id="article"><p> 第一段 </p><p class="readinline">广告</p><p> 第二段 &amp; 后续 </p><!--over--></div>';
  check('original 1321 rule',sourceRuleText(html,source.ruleContent.content),'第一段\n第二段 & 后续');
  check('captures and limit',JSON.stringify(runSourceJavaScript('result.split(/(\\d+)/,4)','a1b22c')),JSON.stringify(['a','1','b','22']));
  check('zero-width boundary',JSON.stringify(runSourceJavaScript('result.split(/$/)','a')),JSON.stringify(['a']));
  let rejected=false;try{runSourceJavaScript('result.split("")','x'.repeat(9000))}catch(error){rejected=String(error).includes('限制')}check('output bounded',String(rejected),'true');
 }catch(error){lines.push('FAIL '+String(error));}
 return <View style={{flex:1,paddingTop:70,paddingHorizontal:15}}><Text>{lines.join('\n')}</Text></View>;
}
AppRegistry.registerComponent('apps',()=>App);
