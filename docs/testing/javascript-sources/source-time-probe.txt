import React from 'react';
import {AppRegistry,Text,View} from 'react-native';
import {sourceRuleText} from './src/core/sourceRule';
function App(){
 const cases=[
  [1704067200000,0,'1 1 +0000 Z Z Z'],
  [1704067200000,19800000,'1 1 +0530 +05 +0530 +05:30'],
  [1704067200000,-12600000,'365 5 -0330 -03 -0330 -03:30'],
  [1704067200000,8,'1 1 +0000 +00 +0000 +00:00'],
  [1704067200000,1234567,'1 1 +0020 +00 +0020 +00:20']
 ];
 const lines=['iOS JavaScript source time'];
 for(const [time,offset,expected] of cases){try{
  const value=sourceRuleText('',`@js:java.timeFormatUTC(${time},"D F Z X XX XXX",${offset})`);
  lines.push((value===expected?'PASS ':'FAIL ')+offset+' '+value);
 }catch(error){lines.push('FAIL '+offset+' '+String(error));}}
 return <View style={{flex:1,paddingTop:70,paddingHorizontal:15}}><Text>{lines.join('\n')}</Text></View>;
}
AppRegistry.registerComponent('apps',()=>App);
