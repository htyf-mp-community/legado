import React,{useEffect,useRef,useState} from 'react';

import {WebView} from 'react-native-webview';
import {HttpServer} from 'react-native-nitro-http-server';
import {startLocalServer,stopLocalServer,localServerState,acquireLocalWorkerPage} from './src/services/localServerRuntime';
const html=`<!doctype html><script>try{const w=new Worker(URL.createObjectURL(new Blob(['onmessage=()=>postMessage(typeof SharedArrayBuffer)'],{type:'text/javascript'})));w.onmessage=e=>{window.ReactNativeWebView.postMessage(e.data);w.terminate()};w.postMessage(1)}catch(e){window.ReactNativeWebView.postMessage('error '+String(e))}</script>`;
function App(){const [lines,setLines]=useState(['iOS native shared listener']),[url,setUrl]=useState('');const resolve=useRef(null);useEffect(()=>{(async()=>{const log=(name,ok)=>setLines(x=>[...x,(ok?'PASS ':'FAIL ')+name]);let lease,external;try{const port=await startLocalServer('web',0,async()=>({statusCode:200,body:'public-web'}));const root='http://127.0.0.1:'+port;lease=await acquireLocalWorkerPage(html);log('worker shares web port',new URL(lease.url).port===String(port));log('public route remains',await (await fetch(root+'/books')).text()==='public-web');const page=await fetch(lease.url);log('isolation headers',page.headers.get('cross-origin-opener-policy')==='same-origin'&&page.headers.get('cross-origin-embedder-policy')==='require-corp');log('worker POST rejected',(await fetch(lease.url,{method:'POST'})).status===405);const ready=new Promise(r=>{resolve.current=r});setUrl(lease.url);log('WebView shared buffer',(await Promise.race([ready,new Promise(r=>setTimeout(()=>r('timeout'),4000))]))==='function');setUrl('');await stopLocalServer('web');log('public stop retains worker',(await localServerState('web',0)).running===false&&(await fetch(lease.url)).status===200);await lease.release();lease=undefined;const prior=await startLocalServer('web',0,async()=>({statusCode:200,body:'before-recovery'}));const kept=await acquireLocalWorkerPage(html);const killer=new HttpServer();if(await killer.isRunning())await killer.stop();await startLocalServer('web',prior,async()=>({statusCode:200,body:'recovered'}));log('manual recovery keeps worker',await (await fetch('http://127.0.0.1:'+prior+'/')).text()==='recovered'&&(await fetch(kept.url)).status===200);await kept.release();await stopLocalServer('web');external=new HttpServer();const extPort=await external.start(0,async()=>({statusCode:200,body:'foreign'}),{host:'127.0.0.1'});let refused=false;try{await acquireLocalWorkerPage('page')}catch(e){refused=true}log('external owner preserved',refused&&await (await fetch('http://127.0.0.1:'+extPort+'/')).text()==='foreign');await external.stop();external=undefined;}catch(e){setLines(x=>[...x,'FAIL '+String(e)])}finally{if(lease)await lease.release().catch(()=>{});if(external)await external.stop().catch(()=>{});}})()},[]);return <View style={{flex:1,paddingTop:70,paddingHorizontal:15}}><Text>{lines.join('\n')}</Text>{url?<WebView style={{height:100,flex:0}} source={{uri:url}} originWhitelist={['*']} onMessage={e=>resolve.current?.(e.nativeEvent.data)}/>:null}</View>}
AppRegistry.registerComponent('apps',()=>App);
