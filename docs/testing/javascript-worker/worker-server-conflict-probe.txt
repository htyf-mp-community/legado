import React,{useEffect,useState} from 'react';
import {AppRegistry,Text,View} from 'react-native';
import {HttpServer} from 'react-native-nitro-http-server';
function App(){const [result,setResult]=useState('testing isolated servers');useEffect(()=>{(async()=>{const lines=[],a=new HttpServer(),b=new HttpServer();const get=async port=>{const c=new AbortController(),t=setTimeout(()=>c.abort(),2000);try{const r=await fetch('http://127.0.0.1:'+port+'/',{signal:c.signal});return await r.text()}catch(e){return 'error '+String(e)}finally{clearTimeout(t)}};try{const port1=await a.start(0,async()=>({statusCode:200,body:'first'}),{host:'127.0.0.1'});lines.push('first before: '+await get(port1));let port2=0;try{port2=await Promise.race([b.start(0,async()=>({statusCode:200,body:'second'}),{host:'127.0.0.1'}),new Promise((_,reject)=>setTimeout(()=>reject(new Error('start timeout')),3000))]);lines.push('second start: '+port2)}catch(e){lines.push('second start failed: '+String(e))}lines.push('first after: '+await get(port1));if(port2>0)lines.push('second after: '+await get(port2));lines.push('first and second same port: '+(port1===port2));}catch(e){lines.push('probe failed: '+String(e))}finally{try{await a.stop()}catch(e){}try{await b.stop()}catch(e){}setResult(lines.join('\n'))}})()},[]);return <View style={{paddingTop:80,paddingHorizontal:15}}><Text>{result}</Text></View>}
AppRegistry.registerComponent('apps',()=>App);
